//type the following code in the nano editor and save it
public class filename{
    public static void main(String args[]){
        long st=System.nanoTime();
        long sum=0;
        for(long i=0;i<1000000000L;i++){
            sum+=i;
        }
        long et=System.nanoTime();
        double ds=(et-st)/1000000000.0;
        System.out.println("sum is "+sum);
        System.out.println("Execution Time "+ds+" seconds");
    }
}

// EXP 6 Transfer of files from one Virtual Machine to another

// Step 1 Configure VirtualBox Network: Set both VMs’ network adapter to "Host-only Adapter” in adapter 1 and NAT in adapter 2(for installing packages) so they are on the same subnet.
// Go to settings -> network -> adapter 1 -> Host only adapter and Adapter2 -> NAT
// Step 2 Check IP Addresses
//On both ubuntu as well as kali ip a (check ip) Ensure both are in same subnet (eg 192.168.56.103/24 ubunttu and 192.168.56.104 kali) both in same subnet 192.168.56.0/24
ip a // check ip address of both vm ubuntu and kali linux

// Step 3 Set	up	NFS	Server	on	Ubuntu	VM(Here	ubuntu	act	as	server), 
sudo apt update

// Step 4  Install	NFS	server
sudo apt install nfs-kernel-server -y

// Step 5 Create a directory to share
sudo mkdir -p /srv/nfs/rakshna 

// Step 6 Set	permissions
sudo chown nobody:nogroup /srv/nfs/rakshna
chmod 777 /srv/nfs/rakshna

// Step 7 Edit	exports	/ile	to	allow	Kali	IP	access
sudo nano /etc/exports

// Step 8 Copy-paste exactly this line into the nano file: (in the place of <kali ip> write the kali linux vm ip address)
/srv/nfs/sharedfolder <kali ip>(rw,sync,no_subtree_check,no_root_squash)

// Step 9 Export and restart NFS server:
sudo exportfs -a
sudo systemctl restart nfs-kernel-server

// Step 10 Allow/disable the firewall for the kali linux (in the place of <kali ip> write the kali linux vm ip address)
sudo ufw allow from <kali ip> to any port nfs

// Step 11 Set up NFS Client on Kali VM (open kali linux terminal) (Here kali act as client)
sudo apt update
sudo apt install nfs-common -y

// Step 12 Create mount point:
sudo mkdir -p /mnt/nfs/rakshna

// Step 13 Mount the NFS share:
sudo mount <ubuntu ip>:/srv/nfs/rakshna /mnt/nfs/rakshna
// Step 14 Verify the mount:
df -h | grep nfs
ls -l /mnt/nfs/rakshna 

// Step 15 Transfer files from Ubuntu to Kali

// EXP 8 Network Virtualization – OpenVSwitch and Linux bridge
//Linux Bridge:
//Step 1 Check the current network configuration.
ifconfig

//Step 2 Install and configure necessary Bridge Utilities
sudo apt install bridge-utils
sudo modprobe bridge
sudo modprobe br_netfilter

//Step 3 Edit the netplan configuration file and specify the configuration.
sudo nano /etc/netplan/01-netcfg.yaml
//Add the following configuration:
network:
  version: 2
  renderer: networkd
  ethernets:
    eno2:
      dhcp4: no
  bridges:
    br-cloud:
      interfaces: [eno2]
      addresses: [10.0.2.15/24]
      gateway4: 10.0.2.255
      nameservers:
        addresses: [8.8.8.8, 1.1.1.1]

// Step 4 Apply the new netplan configuration and restart the network.
sudo systemctl restart networking

//Step 5 Verify network interfaces and bridge
ip a

// Step 6 Verify routing table
route -n
brctl show

// Open Virtual Switch:
// Step 8 Enter root user
sudo -s

// Step 9 Create two namespaces
ip netns add VRF1
ip netns add VRF2
// Check them:
ip netns list

// Step 10 Create virtual Ethernet pairs
ip link add veth1 type veth peer name veth2 // 1st pair
ip link add veth3 type veth peer name veth4 // 2nd pair

// Step 11 Check the virtual Ethernet links
ip link

// Step 12 Assign veth ports to namespaces
ip link set veth1 netns VRF1
ip link set veth3 netns VRF2
ip netns exec VRF1 ip link // check veth1 in VRF1
ip netns exec VRF2 ip link // check veth3 in VRF2 

// Step 13 Assign IP addresses
ip netns exec VRF1 ifconfig veth1 10.10.10.1/24 up
ip netns exec VRF2 ifconfig veth3 10.10.10.2/24 up
ip netns exec VRF2 ifconfig // check the congifuration 
ip netns exec VRF1 ifconfig // check the congifuration

// Step 14 Check routing tables
ip netns exec VRF2 route -n
route -n //Check the host routing table
ip netns exec VRF1 route -n

// Step 15 Install Open vSwitch
sudo apt install openvswitch-switch openvswitch-common -y
ovs-vsctl --version
sudo systemctl status openvswitch-switch // check the service

// Step 16 Create the OVS bridge
ovs-vsctl add-br vSwitch1
ovs-vsctl show // check configuration
ovs-vsctl add-port vSwitch1 veth2
ovs-vsctl add-port vSwitch1 veth4
ovs-vsctl show

// Step 17 Bring virtual Ethernet ports UP
ifconfig veth2 up
ifconfig veth4 up
ifconfig vSwitch1 10.10.10.3/24 up

// Step 18 Test connectivity
ip netns exec VRF1 ping 10.10.10.2

// Step 19 Enable Spanning Tree Protocol
ovs-vsctl set bridge vSwitch1 stp_enable=true
ovsdb-client dump

// Step 20 Check MAC addresses
ovs-appctl fdb/show vSwitch1

ip netns exec VRF1 ifconfig
