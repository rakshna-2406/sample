nano filename.java

//type the following code in the nano editor and save it
public class filename{
    public static void main(String args[]){
        long st=System.nanoTime();
        long sum=0;
        for(long i=0;i<1000000000L;i++){
            sum+=i;
        }
        long et=System.nanoTime();
        double ds=(et-st)/1000000000.0;
        System.out.println("sum is "+sum);
        System.out.println("Execution Time "+ds+" seconds");
    }
}

javac filename.java
java filename
